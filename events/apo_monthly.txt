# AI Pop Optimizer - events (namespace is required)
namespace = apo

# Monthly pulse: iterate AI colonies (fired from on_actions)
apo.1 = {
	hidden = yes
	is_triggered_only = yes

	immediate = {
		every_country = {
			limit = { apo_is_ai_country = yes }
			every_owned_colony = {
				apo_monthly_planet_check = yes
			}
		}
	}
}

# Yearly recalculation of AI economy compensation
apo.2 = {
	hidden = yes
	is_triggered_only = yes

	immediate = {
		every_country = {
			limit = { apo_is_ai_country = yes }
			apo_recalculate_ai_compensation = yes
		}
	}
}
