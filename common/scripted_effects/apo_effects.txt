# AI Pop Optimizer - scripted effects (sintaksis proveren po 4.5.1)

# Ezhemesyachnaya proverka kolonii II: esli popov > cap -> zamorozit rost,
# esli nizhe cap -> snyat zamorozku (oshibka staryh modov: modifikator
# veshalsya navsegda i II navsegda perestaval rasti).
# Ispolzuem every_owned_colony (aktualnyy iterator 4.x,
# every_planet deprecated).
apo_monthly_planet_check = {
	if = {
		limit = {
			owner = { apo_is_ai_country = yes }
		}
		if = {
			limit = { apo_is_over_pop_cap = yes }
			add_modifier = { modifier = apo_pop_growth_freeze days = -1 }
		}
		else = {
			remove_modifier = apo_pop_growth_freeze
		}
	}
}

# Pereschyot kompensatsii ekonomiki II (raz v god).
# num_owned_colonies - proverennyy trigger 4.5 (num_planets ne suschestvuet).
apo_recalculate_ai_compensation = {
	remove_modifier = apo_ai_econ_comp_1
	remove_modifier = apo_ai_econ_comp_2
	remove_modifier = apo_ai_econ_comp_3
	if = {
		limit = { num_owned_colonies >= 20 }
		add_modifier = { modifier = apo_ai_econ_comp_3 days = -1 }
	}
	else_if = {
		limit = { num_owned_colonies >= 10 }
		add_modifier = { modifier = apo_ai_econ_comp_2 days = -1 }
	}
	else_if = {
		limit = { num_owned_colonies >= 3 }
		add_modifier = { modifier = apo_ai_econ_comp_1 days = -1 }
	}
}
