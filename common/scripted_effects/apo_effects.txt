# AI Pop Optimizer — scripted effects

# Ежемесячная проверка планеты ИИ: если попов > cap → заморозить рост,
# если ниже cap → снять заморозку (распространённая ошибка старых модов:
# модификатор вешался навсегда и ИИ навсегда переставал расти)
apo_monthly_planet_check = {
	if = {
		limit = {
			owner = { apo_is_ai_country = yes }
		}
		if = {
			limit = { apo_is_over_pop_cap = yes }
			add_modifier = { modifier = apo_pop_growth_freeze days = -1 }
		}
		else = {
			remove_modifier = apo_pop_growth_freeze
		}
	}
}

# Пересчёт компенсации экономики ИИ (раз в год, чтобы не грузить тик).
# Ключевая идея мода: ИИ с ограниченными попами НЕ слабеет —
# недобор производства компенсируется тировыми модификаторами.
apo_recalculate_ai_compensation = {
	if = {
		limit = { apo_is_ai_country = yes }
		remove_modifier = apo_ai_econ_comp_1
		remove_modifier = apo_ai_econ_comp_2
		remove_modifier = apo_ai_econ_comp_3
		if = {
			limit = { num_planets >= 20 }
			add_modifier = { modifier = apo_ai_econ_comp_3 days = -1 }
		}
		else_if = {
			limit = { num_planets >= 10 }
			add_modifier = { modifier = apo_ai_econ_comp_2 days = -1 }
		}
		else_if = {
			limit = { num_planets >= 3 }
			add_modifier = { modifier = apo_ai_econ_comp_1 days = -1 }
		}
	}
}
